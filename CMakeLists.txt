cmake_minimum_required(VERSION 3.21)
project(VoiceMorphDSP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(voicemorph_dsp INTERFACE)
target_include_directories(voicemorph_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_voice tests/test_voice.cpp)
 target_link_libraries(test_voice PRIVATE voicemorph_dsp)
 add_test(NAME voice_foundation COMMAND test_voice)
 add_executable(test_ir tests/test_ir.cpp)
 target_link_libraries(test_ir PRIVATE voicemorph_dsp)
 add_test(NAME ir_reference COMMAND test_ir)
 add_executable(test_wav tests/test_wav.cpp)
 target_link_libraries(test_wav PRIVATE voicemorph_dsp)
 add_test(NAME wav_import COMMAND test_wav)
 add_executable(test_ir_matrix tests/test_ir_matrix.cpp)
 target_link_libraries(test_ir_matrix PRIVATE voicemorph_dsp)
 add_test(NAME ir_channel_matrix COMMAND test_ir_matrix)
 add_executable(test_partitioned_ir tests/test_partitioned_ir.cpp)
 target_link_libraries(test_partitioned_ir PRIVATE voicemorph_dsp)
 add_test(NAME partitioned_ir COMMAND test_partitioned_ir)
 add_executable(test_fft_stereo tests/test_fft_stereo.cpp)
 target_link_libraries(test_fft_stereo PRIVATE voicemorph_dsp)
 add_test(NAME fft_stereo_matrix COMMAND test_fft_stereo)
 add_executable(test_ir_resampler tests/test_ir_resampler.cpp)
 target_link_libraries(test_ir_resampler PRIVATE voicemorph_dsp)
 add_test(NAME ir_resampler COMMAND test_ir_resampler)
 add_executable(test_ir_stage tests/test_ir_stage.cpp)
 target_link_libraries(test_ir_stage PRIVATE voicemorph_dsp)
 add_test(NAME ir_stage_alignment COMMAND test_ir_stage)
 add_executable(test_meter_units tests/test_meter_units.cpp)
 target_link_libraries(test_meter_units PRIVATE voicemorph_dsp)
 add_test(NAME meter_dbfs_conversion COMMAND test_meter_units)
 add_executable(test_ir_pipeline tests/test_ir_pipeline.cpp)
 target_link_libraries(test_ir_pipeline PRIVATE voicemorph_dsp)
 add_test(NAME wav_to_ir_pipeline COMMAND test_ir_pipeline)
 # CMake Release defines NDEBUG; preserve assert-based test execution.
 if(MSVC)
  foreach(t test_voice test_ir test_wav test_ir_matrix test_partitioned_ir test_fft_stereo test_ir_resampler test_ir_stage test_meter_units test_ir_pipeline)
   target_compile_options(${t} PRIVATE /UNDEBUG)
  endforeach()
 else()
  foreach(t test_voice test_ir test_wav test_ir_matrix test_partitioned_ir test_fft_stereo test_ir_resampler test_ir_stage)
   target_compile_options(${t} PRIVATE -UNDEBUG)
  endforeach()
 endif()
endif()

option(BUILD_VST3 "Build the actual Steinberg VST3 wrapper" OFF)
if(BUILD_VST3)
 add_subdirectory(vst)
endif()
