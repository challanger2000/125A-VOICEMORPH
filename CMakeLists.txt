cmake_minimum_required(VERSION 3.21)
project(VoiceMorphDSP VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(voicemorph_dsp INTERFACE)
target_include_directories(voicemorph_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_voice tests/test_voice.cpp)
 target_link_libraries(test_voice PRIVATE voicemorph_dsp)
 add_test(NAME voice_foundation COMMAND test_voice)
 add_executable(test_ir tests/test_ir.cpp)
 target_link_libraries(test_ir PRIVATE voicemorph_dsp)
 add_test(NAME ir_reference COMMAND test_ir)
 add_executable(test_wav tests/test_wav.cpp)
 target_link_libraries(test_wav PRIVATE voicemorph_dsp)
 add_test(NAME wav_import COMMAND test_wav)
 add_executable(test_ir_matrix tests/test_ir_matrix.cpp)
 target_link_libraries(test_ir_matrix PRIVATE voicemorph_dsp)
 add_test(NAME ir_channel_matrix COMMAND test_ir_matrix)
endif()

option(BUILD_VST3 "Build the actual Steinberg VST3 wrapper" OFF)
if(BUILD_VST3)
 add_subdirectory(vst)
endif()
