# Native GUI integration is not part of the headless DSP unit suite: this
# executes the *packaged* VST3 through the actual VST3 hosting loader and
# exercises Windows HWND attach/focus/detach/reopen. It must run on Windows.
if(BUILD_VST3 AND WIN32 AND BUILD_TESTING)
 # Workflow SDK location arrives as a Windows path containing backslashes.
 # Normalize before creating the test target's source-file argument.
 file(TO_CMAKE_PATH "${VST3_SDK_ROOT}" VM_SDK_CMAKE_PATH)
 add_executable(test_vm_editor_lifecycle
  tests/test_vm_editor_lifecycle.cpp
  "${VM_SDK_CMAKE_PATH}/public.sdk/source/vst/hosting/module_win32.cpp"
  "${VM_SDK_CMAKE_PATH}/public.sdk/source/common/memorystream.cpp")
 target_link_libraries(test_vm_editor_lifecycle PRIVATE
  sdk_hosting sdk_common base pluginterfaces user32 ole32)
 target_compile_definitions(test_vm_editor_lifecycle PRIVATE
  _CRT_SECURE_NO_WARNINGS NOMINMAX UNICODE _UNICODE)
 target_compile_options(test_vm_editor_lifecycle PRIVATE /UNDEBUG)
 add_dependencies(test_vm_editor_lifecycle VoiceMorph)
endif()

# Same module-level audio/state regression on Windows and Linux. Windows builds
# include the real GUI; VM_WITH_GUI=OFF permits explicit headless Linux QA only.
if(BUILD_VST3 AND BUILD_TESTING)
 file(TO_CMAKE_PATH "${VST3_SDK_ROOT}" VM_HOST_SDK_PATH)
 if(WIN32)
  set(vm_host_module "${VM_HOST_SDK_PATH}/public.sdk/source/vst/hosting/module_win32.cpp")
 elseif(UNIX AND NOT APPLE)
  set(vm_host_module "${VM_HOST_SDK_PATH}/public.sdk/source/vst/hosting/module_linux.cpp")
 endif()
 if(DEFINED vm_host_module)
  add_executable(test_vm_host tests/test_vm_host.cpp "${vm_host_module}"
   "${VM_HOST_SDK_PATH}/public.sdk/source/common/memorystream.cpp")
  target_link_libraries(test_vm_host PRIVATE sdk_hosting sdk_common base pluginterfaces voicemorph_dsp)
  if(WIN32)
   target_link_libraries(test_vm_host PRIVATE user32 ole32)
   target_compile_definitions(test_vm_host PRIVATE NOMINMAX UNICODE _UNICODE)
  else()
   target_link_libraries(test_vm_host PRIVATE ${CMAKE_DL_LIBS})
  endif()
  add_dependencies(test_vm_host VoiceMorph)
  add_test(NAME actual_vst3_audio_state_recall COMMAND test_vm_host "$<TARGET_PROPERTY:VoiceMorph,SMTG_PLUGIN_PACKAGE_PATH>")
 endif()
endif()
